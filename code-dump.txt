    if (!username || !password)
      return Response.json({ success: false, error: "نام کاربری و رمز را وارد کنید" }, { status: 400 });

    const db = getDB();
    const user = await db
      .prepare("SELECT * FROM users WHERE (username = ?1 OR email = ?1) AND active = 1")
      .bind(username.trim())
      .first();

    // پیام یکسان برای «کاربر نیست» و «رمز غلط» (کسی نتواند حدس بزند کدام حساب وجود دارد)

      return Response.json({ success: false, error: "نام کاربری یا رمز عبور اشتباه است" }, { status: 401 });

    await createSession(user);
    return Response.json({ success: true, role: user.role, home: HOME_BY_ROLE[user.role] || "/" });
  } catch {
    return Response.json({ success: false, error: "خطا در برقراری ارتباط" }, { status: 500 });
  }
}

// مبلغ: ارقام فارسی را قبول می‌کند، خالی = null، نامعتبر = "bad"
function money(v: unknown): number | null | "bad" {
  const s = toEnDigits(String(v ?? "")).replace(/[,،\s]/g, "");
  if (!s) return null;
  if (!/^\d{1,10}$/.test(s)) return "bad";
  return Number(s);
}

type Entry = { none: boolean; items: string[]; when: string; note: string };

// سوابق را تمیز می‌کند و مطمئن می‌شود به هر شش سؤال جواب داده شده
function cleanHistory(h: unknown): { data?: Record<string, Entry>; error?: string } {
  const src = (h && typeof h === "object" ? h : {}) as Record<string, any>;
  const data: Record<string, Entry> = {};
  for (const q of HISTORY_QUESTIONS) {
    const v = src[q.key] ?? {};
    const none = v.none === true;
    const items: string[] =
      none || !Array.isArray(v.items)
        ? []
        : v.items.slice(0, 20).map((s: unknown) => normText(s).slice(0, 40)).filter(Boolean);
    if (!none && items.length === 0) return { error: `به سؤال «${q.title}» پاسخ داده نشده` };
    data[q.key] = {
      none,
      items,
      when: none || !q.hasWhen ? "" : normText(v.when).slice(0, 40),
      note: normText(v.note).slice(0, 30),
    };
  }
  return { data };
}

    .replace(/ي/g, "ی")
    .replace(/ك/g, "ک")
    .replace(/\s+/g, " ")
    .trim();
}

// هش رمز با PBKDF2 (سازگار با Cloudflare؛ سقف مجاز تکرار ۱۰۰۰۰۰ است)
const enc = new TextEncoder();
const toHex = (b: ArrayBuffer | Uint8Array) =>
  [...new Uint8Array(b as ArrayBuffer)].map((x) => x.toString(16).padStart(2, "0")).join("");
const fromHex = (h: string) =>
  new Uint8Array((h.match(/.{2}/g) || []).map((x) => parseInt(x, 16)));

async function derive(password: string, salt: Uint8Array) {
  const key = await crypto.subtle.importKey("raw", enc.encode(password), "PBKDF2", false, ["deriveBits"]);
  return crypto.subtle.deriveBits(
    { name: "PBKDF2", hash: "SHA-256", salt: salt as BufferSource, iterations: 100000 },
    key,
    256
  );
}

export async function hashPassword(password: string) {
  const salt = crypto.getRandomValues(new Uint8Array(16));
  return `${toHex(salt)}:${toHex(await derive(password, salt))}`;
}

export async function verifyPassword(password: string, stored: string) {
  const [s, h] = (stored || "").split(":");
  if (!s || !h) return false;
  return toHex(await derive(password, fromHex(s))) === h;
}

import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";
import { getEnv } from "./db";

export const COOKIE = "auth_token";

export type SessionUser = { uid: number; role: string; name: string };

function secretKey() {
  const s = getEnv().JWT_SECRET;
  if (!s || s.length < 32) throw new Error("JWT_SECRET تنظیم نشده یا کوتاه است (حداقل ۳۲ کاراکتر)");
  return new TextEncoder().encode(s);
}

export async function createSession(user: { id: number; role: string; full_name: string }) {
  const token = await new SignJWT({ uid: user.id, role: user.role, name: user.full_name })
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime("12h")
    .sign(secretKey());
  (await cookies()).set(COOKIE, token, {
    httpOnly: true,
    sameSite: "lax",
    path: "/",
    maxAge: 60 * 60 * 12,
    secure: process.env.NODE_ENV === "production",
  });
}

export async function getSession(): Promise<SessionUser | null> {
  const token = (await cookies()).get(COOKIE)?.value;
  if (!token) return null;
  try {
    const { payload } = await jwtVerify(token, secretKey());
    const uid = Number(payload.uid);
    // وضعیت و نقش را از دیتابیس می‌گیریم تا غیرفعال‌سازی فوری اعمال شود
    const row = await getEnv()
      .DB.prepare("SELECT role, full_name, active FROM users WHERE id = ?")
      .bind(uid)
      .first();
    if (!row || !row.active) return null;
    return { uid, role: String(row.role), name: String(row.full_name || "") };
  } catch {
    return null;
  }
}

export async function destroySession() {
  (await cookies()).delete(COOKIE);
}
